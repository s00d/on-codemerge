Client-only editor — [Meta SSR](/integrate/basics/meta-ssr). UI: [Vue 3](/integrate/frontend/vue3) via `@codemerge/integrate/nuxt` (re-exports `/vue` + `clientOnlyHint()`).

```vue
<script setup lang="ts">
import { ref } from 'vue';
import { CodeMergeEditor } from '@codemerge/integrate/nuxt';

const html = ref('<p></p>');

function onChange(value: string) {
  html.value = value;
}
</script>

<template>
  <ClientOnly>
    <CodeMergeEditor :value="html" format="html" @change="onChange" />
  </ClientOnly>
</template>
```

Packs / upload: [Host config](/integrate/basics/host-config). Load/save: [Persistence](/integrate/basics/persistence).
